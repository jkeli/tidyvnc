cmake_minimum_required(VERSION 3.15)
project(TigerVNC_FLTK LANGUAGES C CXX)

# This is a separate dependency build: the main project never downloads code.
include(ExternalProject)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
option(FLTK_SHARED "Build shared FLTK as well as static libraries" OFF)
set(FLTK_ARCHIVE
  "https://github.com/fltk/fltk/releases/download/release-1.4.5/fltk-1.4.5-source.tar.gz"
  CACHE STRING "FLTK archive URL (or local verified archive)")
ExternalProject_Add(fltk
  URL "${FLTK_ARCHIVE}"
  URL_HASH SHA256=eede1fb2b8e9c2e581e77082e15252145855c79aad30070ee3b24aabe2f926f1
  CMAKE_ARGS
    -DCMAKE_INSTALL_PREFIX:PATH=${CMAKE_INSTALL_PREFIX}
    -DCMAKE_BUILD_TYPE:STRING=${CMAKE_BUILD_TYPE}
    -DCMAKE_C_COMPILER:FILEPATH=${CMAKE_C_COMPILER}
    -DCMAKE_CXX_COMPILER:FILEPATH=${CMAKE_CXX_COMPILER}
    -DCMAKE_OSX_ARCHITECTURES:STRING=${CMAKE_OSX_ARCHITECTURES}
    -DCMAKE_OSX_DEPLOYMENT_TARGET:STRING=${CMAKE_OSX_DEPLOYMENT_TARGET}
    -DFLTK_BUILD_SHARED_LIBS:BOOL=${FLTK_SHARED}
    -DFLTK_BACKEND_WAYLAND:BOOL=OFF
    -DFLTK_BUILD_GL:BOOL=OFF
    -DFLTK_BUILD_FLUID:BOOL=OFF
    -DFLTK_BUILD_FLTK_OPTIONS:BOOL=OFF
    -DFLTK_BUILD_TEST:BOOL=OFF
    -DFLTK_BUILD_EXAMPLES:BOOL=OFF
    -DFLTK_BUILD_FORMS:BOOL=OFF
    -DFLTK_BUILD_HTML_DOCS:BOOL=OFF
    -DFLTK_BUILD_PDF_DOCS:BOOL=OFF)
